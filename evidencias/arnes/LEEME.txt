  java -m jdk.compiler/com.sun.tools.javac.Main -d /tmp/cls $(find evidencias/arnes -name "*.java") \
    src/main/java/pe/edu/utp/piuravigila/{model,dto,dao,service}/*.java \
    src/main/java/pe/edu/utp/piuravigila/config/{RelojConfig,FotoStorageDisco}.java src/test/java/pe/edu/utp/piuravigila/service/*.java
  java -cp /tmp/cls Runner pe.edu.utp.piuravigila.service.ReporteServiceTest pe.edu.utp.piuravigila.service.ControlAntiSpamTest pe.edu.utp.piuravigila.service.OtrosServiciosTest pe.edu.utp.piuravigila.service.FotoStorageDiscoTest
